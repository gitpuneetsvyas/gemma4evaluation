'Developer', 'Tester'

-- 2. Users Table
CREATE TABLE Users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(255),
    role_id INTEGER NOT NULL REFERENCES Roles(role_id),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Projects Table
CREATE TABLE Projects (
    project_id SERIAL PRIMARY KEY,
    project_name VARCHAR(255) NOT NULL,
    description TEXT,
    status VARCHAR(50) NOT NULL DEFAULT 'Planning', -- e.g., Planning, In P[1D[K
Progress, Completed, Blocked
    priority VARCHAR(50) NOT NULL DEFAULT 'Medium', -- e.g., Low, Medium, H[1D[K

    start_date DATE,

    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. Tasks Table
CREATE TABLE Tasks (
    task_id SERIAL PRIMARY KEY,
    project_id INTEGER NOT NULL REFERENCES Projects(project_id),
    task_name VARCHAR(255) NOT NULL,
    description TEXT,
    status VARCHAR(50) NOT NULL DEFAULT 'To Do', -- e.g., To Do, In Progres[7D[K
Progress, Review, Done
    priority VARCHAR(50) NOT NULL DEFAULT 'Medium',
    assigned_to_user_id INTEGER REFERENCES Users(user_id),
    due_date TIMESTAMP WITH TIME ZONE,

    dependency_task_id INTEGER REFERENCES Tasks(task_id), -- For complex wo[2D[K

    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. Comments Table
CREATE TABLE Comments (
    comment_id SERIAL PRIMARY KEY,
    task_id INTEGER NOT NULL REFERENCES Tasks(task_id),
