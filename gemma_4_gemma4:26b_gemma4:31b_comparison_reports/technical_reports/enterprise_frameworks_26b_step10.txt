// order.gateway.ts (NestJS + Socket.io)
@WebSocketGateway({ cors: true, namespace: 'orders' })
export class OrderGateway implements OnModuleInit {
  @WebSocketServer() server: Server;

  // In a real scenario, we use a Redis Adapter to allow scaling
  // and listen to RabbitMQ/Kafka messages
  constructor(private readonly amqpService: AmqpService) {}

  async onModuleInit() {
    // Listening to the Event Mesh to push updates to the UI
    await this.amqpService.consume('modern.orders.exchange', (msg) => {
      this.server.emit('orderUpdate', msg);
    });
  }

  @SubscribeMessage('subscribeToOrder')
  handleSubscription(client: Socket, orderId: string) {
    client.join(`order_${orderId}`);
    return { status: 'Subscribed to real-time updates' };
  }
}

// order.service.ts (Prisma Integration)
@Injectable()

      include: { customer: true }
